// Rail Priority - flag: while this script runs, the jam guard in this save is
// 180 s (a held freight train goes if the same passenger train has not moved
// for that long). It does nothing else; the plugin looks for it.

include("SOVIETInstructions.txt");

defineVariable(char, bRun);

defineFunction(main, void)
{
	bRun = 1;
	while (bRun)
	{
		Script_Sleep(60.0);
	}
	end();
}
