// Rail Priority - flag: while this script runs, the maximum wait in this save is
// 60 s (a freight train waits at most that long for passenger trains). It does nothing else; the plugin looks for it.

include("SOVIETInstructions.txt");

defineVariable(char, bRun);

defineFunction(main, void)
{
	bRun = 1;
	while (bRun)
	{
		Script_Sleep(60.0);
	}
	end();
}
