// Rail Priority - button: the game decides the order at signals (OFF).
// Starts the OFF flag script, unless it runs already, and says so in the
// player's language (text ids 1957001 / 1957003 are served by wrsr_rail_mod.dll).

include("SOVIETInstructions.txt");

defineVariable(char, bOff);
defineVariable(vec3, vWhere);

defineFunction(main, void)
{
	Script_Exists("RailPriority_OffFlag.txt", bOff);
	if (!bOff) { Script_StartNew("RailPriority_OffFlag.txt"); }
	Notification_CreateNewLangIDPic(1957001, 1957003, "", vWhere);
	end();
}
